cmake_minimum_required(VERSION 3.16)
project(cpp_autograd_engine LANGUAGES CXX)
add_library(autograd INTERFACE)
target_include_directories(autograd INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(autograd INTERFACE cxx_std_17)
add_executable(scalar_demo examples/scalar_demo.cpp)
target_link_libraries(scalar_demo PRIVATE autograd)
add_executable(xor_demo examples/xor_demo.cpp)
target_link_libraries(xor_demo PRIVATE autograd)
add_executable(model_io_demo examples/model_io_demo.cpp)
target_link_libraries(model_io_demo PRIVATE autograd)
add_executable(tensor_demo examples/tensor_demo.cpp)
target_link_libraries(tensor_demo PRIVATE autograd)
add_executable(classification_demo examples/classification_demo.cpp)
target_link_libraries(classification_demo PRIVATE autograd)
add_executable(multiclass_demo examples/multiclass_demo.cpp)
target_link_libraries(multiclass_demo PRIVATE autograd)
include(CTest)
if(BUILD_TESTING)
    add_executable(scalar_test tests/scalar_test.cpp)
    target_link_libraries(scalar_test PRIVATE autograd)
    add_test(NAME scalar_gradients COMMAND scalar_test)
    add_executable(mlp_test tests/mlp_test.cpp)
    target_link_libraries(mlp_test PRIVATE autograd)
    add_test(NAME mlp_gradients_and_xor COMMAND mlp_test)
    add_test(NAME xor_demo_convergence COMMAND xor_demo)
    add_executable(model_io_test tests/model_io_test.cpp)
    target_link_libraries(model_io_test PRIVATE autograd)
    add_test(NAME model_serialization COMMAND model_io_test)
    add_executable(tensor_test tests/tensor_test.cpp)
    target_link_libraries(tensor_test PRIVATE autograd)
    add_test(NAME tensor_gradients COMMAND tensor_test)
    add_executable(classification_test tests/classification_test.cpp)
    target_link_libraries(classification_test PRIVATE autograd)
    add_test(NAME classification_loss COMMAND classification_test)
    add_executable(broadcast_test tests/broadcast_test.cpp)
    target_link_libraries(broadcast_test PRIVATE autograd)
    add_test(NAME row_broadcasting COMMAND broadcast_test)
    add_executable(multiclass_test tests/multiclass_test.cpp)
    target_link_libraries(multiclass_test PRIVATE autograd)
    add_test(NAME multiclass_training COMMAND multiclass_test)
    add_test(NAME multiclass_demo_convergence COMMAND multiclass_demo)
endif()
